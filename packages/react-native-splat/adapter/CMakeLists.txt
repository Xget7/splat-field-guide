cmake_minimum_required(VERSION 3.22)
project(sfg_adapter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(SFG_BUILD_TESTS "Build portable host adapter tests" ${PROJECT_IS_TOP_LEVEL})
if(SFG_BUILD_TESTS)
  enable_testing()
  set(SPLATKIT_ENGINE_BUILD_TESTS ON CACHE BOOL "" FORCE)
endif()
if(NOT TARGET splatkit_engine)
  add_subdirectory(../engine/splatkit-engine splatkit-engine)
endif()
add_library(sfg_adapter STATIC src/sfg.cpp)
target_include_directories(sfg_adapter PUBLIC include)
target_link_libraries(sfg_adapter PUBLIC splatkit_engine)
target_compile_options(sfg_adapter PRIVATE
  $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall -Wextra -Wpedantic -Werror>)
if(SFG_BUILD_TESTS)
  add_executable(sfg_adapter_tests tests/SfgTest.cpp)
  target_link_libraries(sfg_adapter_tests PRIVATE sfg_adapter GTest::gtest_main)
  target_compile_options(sfg_adapter_tests PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall -Wextra -Wpedantic -Werror>)
  include(GoogleTest)
  gtest_discover_tests(sfg_adapter_tests TEST_FILTER "-SfgTest.ExpectedGap*")
  foreach(gap OrbitBeforeLoad NearPoleAndSmallRadiusLimits)
    add_test(NAME SfgExpectedGap.${gap}
      COMMAND sfg_adapter_tests --gtest_filter=SfgTest.ExpectedGap${gap})
    set_tests_properties(SfgExpectedGap.${gap} PROPERTIES WILL_FAIL TRUE LABELS expected-api-gap)
  endforeach()
endif()
