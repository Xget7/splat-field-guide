cmake_minimum_required(VERSION 3.22)
project(sfg_metal LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SFG_METAL_BUILD_TESTS "Build host Metal adapter tests" OFF)
set(SPLATKIT_IOS_BUILD_TESTS ${SFG_METAL_BUILD_TESTS} CACHE BOOL "" FORCE)
set(SFG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(SFG_METAL_BUILD_TESTS)
  enable_testing()
endif()
add_subdirectory(../../engine/splatkit-ios splatkit-ios)
add_subdirectory(.. adapter)
add_library(sfg_metal STATIC sfg_metal.mm)
target_include_directories(sfg_metal PUBLIC include)
target_link_libraries(sfg_metal PUBLIC sfg_adapter splatkit_ios)
target_compile_options(sfg_metal PRIVATE -Wall -Wextra -Werror -fobjc-arc)
if(SFG_METAL_BUILD_TESTS)
  add_executable(sfg_metal_tests tests/SfgMetalTest.mm)
  target_include_directories(sfg_metal_tests PRIVATE ../tests)
  target_link_libraries(sfg_metal_tests PRIVATE sfg_metal GTest::gtest_main)
  target_compile_options(sfg_metal_tests PRIVATE -Wall -Wextra -Werror -fobjc-arc)
  include(GoogleTest)
  gtest_discover_tests(sfg_metal_tests)
endif()
